func DrawNO()->void:

	var gap := 20
	var height := 10
	var width := 100
	var endPos := nextDrawPos + Vector2(width / 2.0 - gap, 0)
	# ----
	draw_line(nextDrawPos, endPos, lineColor, lineWidth)
	
	var verticalStart := endPos + Vector2(0, -height) # lineWidth / 2.0  < X >
	var verticalStop :=  verticalStart + Vector2(0, 2 * height)
	# ----|
	draw_line(verticalStart, verticalStop, lineColor, lineWidth)
	
	endPos.x += gap
	verticalStart = endPos + Vector2(0, -height)
	verticalStop =  verticalStart + Vector2(0, 2 * height)
	# ----| |
	draw_line(verticalStart, verticalStop, lineColor, lineWidth)
	
	nextDrawPos = endPos + Vector2(width / 2.0 - gap, 0)
	# ----| |----
	draw_line(endPos, nextDrawPos, lineColor, lineWidth)

func DrawCoil()->void:
	var circleRadius := 10
	# --
	draw_line(nextDrawPos, nextDrawPos + Vector2(20,0), Color.RED, lineWidth)
	nextDrawPos.x += 20 + circleRadius
	# -(
	draw_arc(nextDrawPos, circleRadius, 3 * PI / 2.0, PI / 2.0, 10, Color.WHITE, lineWidth)
	# -( )
	draw_arc(nextDrawPos + Vector2(10, 0), circleRadius, -PI / 2.0, PI / 2.0, 10, Color.WHITE, lineWidth)
	nextDrawPos += Vector2(10 + 10, 0)
	# -( )-|
	draw_line(nextDrawPos, nextDrawPos + Vector2(10,0), Color.RED, lineWidth)
	nextDrawPos.x += 10
	draw_line(nextDrawPos + Vector2(0, -10), nextDrawPos + Vector2(0,10), Color.RED, lineWidth)
